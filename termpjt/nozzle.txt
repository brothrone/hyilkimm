! ==================================================================
!  Term Project : Thermal Stress Analysis of a Rocket Nozzle
!  ANSYS : File > Read Input from... 으로 이 파일 실행
! ==================================================================
/clear
/prep7
/pnum,line,1
/pnum,kp,1

! ==================================================================
!  (a) offset 매크로 : *create ~ *end 사이 내용이 offln.mac 파일로 저장되고,
!      그 다음부터는  offln  이라고 쓰면 이 내용이 실행됨
! ==================================================================
*create,offln,mac
*if,rad,eq,0,then
  dx = x2-x1                         ! 방향벡터
  dy = y2-y1
  lng = sqrt(dx*dx+dy*dy)            ! 길이
  enx = -dy/lng                      ! 90도 돌린 단위 법선벡터
  eny =  dx/lng
  k,kp1,x1+dist*enx,y1+dist*eny
  k,kp2,x2+dist*enx,y2+dist*eny
  l,kp1,kp2
*else
  sc = (rad-dist)/rad                ! 중심 쪽으로 당기는 비율
  k,kp1,xc+(x1-xc)*sc,yc+(y1-yc)*sc
  k,kp2,xc+(x2-xc)*sc,yc+(y2-yc)*sc
  larc,kp1,kp2,kpc,rad-dist
*endif
*end

! ==================================================================
!  (b) wireframe   (단위 : mm, W, K, MPa)
!      축대칭 : Y축 = 노즐 중심축,  X = 반지름,  Y=0 = throat 높이
! ==================================================================
*afun,deg                    ! sin, cos, tan 을 도(degree) 단위로
t   = 30                     ! 벽 두께
rt  = 150                    ! throat 안쪽 반지름
r1  = 200                    ! throat 위쪽 곡률반경
r2  = 50                     ! throat 아래쪽 곡률반경
a1  = 45                     ! 수렴부 각도
a2  = 15                     ! 확산부 각도
yt  = 260                    ! 입구 높이
ye  = -1228.5                ! 출구 높이

xb = rt+r1-r1*cos(a1)        ! R200 원호와 45도 직선이 만나는 점
yb = r1*sin(a1)
xa = xb+(yt-yb)*tan(a1)      ! 입구 안쪽 x (= 327.16)
xd = rt+r2-r2*cos(a2)        ! R50 원호와 15도 직선이 만나는 점
yd = -r2*sin(a2)
xe = xd+(yd-ye)*tan(a2)      ! 출구 안쪽 x (= 477.41)

! ---- 안쪽면 ----
k,1,xa,yt
k,2,xb,yb
k,3,rt,0
k,4,xd,yd
k,5,xe,ye
k,11,rt+r1,0                 ! R200 중심
k,12,rt+r2,0                 ! R50 중심

l,1,2                        ! L1 : 45도 직선
larc,2,3,11,r1               ! L2 : R200
larc,3,4,12,r2               ! L3 : R50
l,4,5                        ! L4 : 15도 직선

! ---- 바깥면 : 안쪽 선 4개를 매크로로 offset ----
dist = t

x1 = xa                      ! L5 : L1(45도 직선) offset
y1 = yt
x2 = xb
y2 = yb
rad = 0
kp1 = 21
kp2 = 22
offln

x1 = xb                      ! L6 : L2(R200) offset -> R170
y1 = yb
x2 = rt
y2 = 0
rad = r1
xc = rt+r1
yc = 0
kpc = 11
kp1 = 23
kp2 = 24
offln

x1 = rt                      ! L7 : L3(R50) offset -> R20
y1 = 0
x2 = xd
y2 = yd
rad = r2
xc = rt+r2
yc = 0
kpc = 12
kp1 = 25
kp2 = 26
offln

x1 = xd                      ! L8 : L4(15도 직선) offset
y1 = yd
x2 = xe
y2 = ye
rad = 0
kp1 = 27
kp2 = 28
offln

nummrg,kp                    ! 이음매에서 겹친 keypoint 합치기 (22=23, 24=25, 26=27)
kdele,11,12                  ! 중심점은 이제 필요 없음

! ---- 바깥 직선의 끝을 입구/출구 높이로 옮기기 ----
! 평행한 두 선(간격 t)의 가로 간격 = t/cos(각도)
kmodif,21,xa+t/cos(a1),yt
kmodif,28,xe+t/cos(a2),ye

! ---- 두께 방향 선 ----
l,1,21                       ! L9  : 입구 끝면
l,2,22                       ! L10
l,3,24                       ! L11
l,4,26                       ! L12
l,5,28                       ! L13 : 출구 끝면
lplot

! ==================================================================
!  (c) 단면 area : 4변짜리 area 4개
! ==================================================================
al,1,10,5,9
al,2,11,6,10
al,3,12,7,11
al,4,13,8,12
aplot

! ==================================================================
!  (d) mapped mesh
! ==================================================================
et,1,plane55
keyopt,1,3,1                 ! 축대칭
mp,kxx,1,1.143e-3            ! W/mm.K

lsel,s,line,,9,13            ! 두께 방향 6등분
lesize,all,,,6
lsel,all
lesize,1,,,34                ! 마주보는 변은 분할 수가 같아야 함
lesize,5,,,34
lesize,2,,,30
lesize,6,,,30
lesize,3,,,4
lesize,7,,,4
lesize,4,,,150
lesize,8,,,150
mshkey,1                     ! mapped mesh
amesh,all
eplot

! ==================================================================
!  (e) 정상상태 열해석
! ==================================================================
lsel,s,line,,1,4             ! 안쪽면 : 연소가스
sfl,all,conv,84e-6,,1073.15
lsel,s,line,,5,8             ! 바깥면 : 공기
sfl,all,conv,30e-6,,298.15
lsel,all

/solu
solve

/post1
set
plnsol,temp

! ==================================================================
!  (f) 열응력 해석
! ==================================================================
/prep7
etchg,tts                    ! PLANE55 -> PLANE182
keyopt,1,3,1                 ! 축대칭
mp,ex,1,208e3                ! MPa
mp,nuxy,1,0.29
mp,alpx,1,1.28e-5
sfldele,all,all              ! 열해석 대류 하중 삭제
ldread,temp,,,,,file,rth     ! 열해석 결과 온도 불러오기
tref,298.15
dl,9,,uy,0                   ! 입구 끝면 축방향 고정

/solu
solve

/post1
set
plnsol,s,eqv
