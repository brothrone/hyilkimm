! ==================================================================
!  Term Project : Thermal Stress Analysis of a Rocket Nozzle
!  ANSYS : File > Read Input from... 로 이 파일 실행
! ==================================================================
/clear
/prep7
/pnum,line,1
/pnum,kp,1

! ==================================================================
!  (a) offset 매크로 : 아래 *create ~ *end 가 offln.mac 파일을 만든다
! ==================================================================
*create,offln,mac
nl_ = arg1
d_  = arg2
ks_ = arg3

! ---- 1) 시작점 (LFRAC = 0) ----
x1_ = lx(nl_,0)
y1_ = ly(nl_,0)
tx_ = lsx(nl_,0)
ty_ = lsy(nl_,0)
tl_ = sqrt(tx_**2+ty_**2)
x1_ = x1_ - d_*ty_/tl_
y1_ = y1_ + d_*tx_/tl_

! ---- 2) 중간점 (LFRAC = 0.5) ----
xm_ = lx(nl_,0.5)
ym_ = ly(nl_,0.5)
tx_ = lsx(nl_,0.5)
ty_ = lsy(nl_,0.5)
tl_ = sqrt(tx_**2+ty_**2)
xm_ = xm_ - d_*ty_/tl_
ym_ = ym_ + d_*tx_/tl_

! ---- 3) 끝점 (LFRAC = 1) ----
x2_ = lx(nl_,1)
y2_ = ly(nl_,1)
tx_ = lsx(nl_,1)
ty_ = lsy(nl_,1)
tl_ = sqrt(tx_**2+ty_**2)
x2_ = x2_ - d_*ty_/tl_
y2_ = y2_ + d_*tx_/tl_

! ---- 4) 직선인지 판정 : 중간점이 시작-끝 현(chord)에서 벗어난 거리 ----
cx_  = x2_-x1_
cy_  = y2_-y1_
cl_  = sqrt(cx_**2+cy_**2)
dev_ = abs(cx_*(ym_-y1_)-cy_*(xm_-x1_))/cl_
tol_ = 1e-6*cl_

! ---- 5) keypoint 만들기 ----
*get,km_,kp,0,num,maxd
*if,ks_,gt,0,then
  ofk1_ = ks_
*else
  ofk1_ = km_+1
  k,ofk1_,x1_,y1_
*endif
ofk2_ = km_+2
k,ofk2_,x2_,y2_

! ---- 6) 선 만들기 ----
*if,dev_,lt,tol_,then
  l,ofk1_,ofk2_
*else
  ! 세 점을 지나는 원의 중심 (xc_,yc_) 과 반지름 r_
  a1_ = x1_**2+y1_**2
  am_ = xm_**2+ym_**2
  a2_ = x2_**2+y2_**2
  dd_ = 2*(x1_*(ym_-y2_)+xm_*(y2_-y1_)+x2_*(y1_-ym_))
  xc_ = (a1_*(ym_-y2_)+am_*(y2_-y1_)+a2_*(y1_-ym_))/dd_
  yc_ = (a1_*(x2_-xm_)+am_*(x1_-x2_)+a2_*(xm_-x1_))/dd_
  r_  = sqrt((x1_-xc_)**2+(y1_-yc_)**2)
  kc_ = km_+3
  k,kc_,xc_,yc_
  larc,ofk1_,ofk2_,kc_,r_
  kdele,kc_
*endif
*get,ofl_,line,0,num,maxd
*end

! ==================================================================
!  (b) 노즐 안쪽면 wireframe  (단위 : mm, W, K, MPa)
!      축대칭 : Y축 = 노즐 중심축,  X = 반지름,  Y=0 = throat 높이
! ==================================================================
*afun,deg
t   = 30         ! 벽 두께
rt  = 150        ! throat 안쪽 반지름
r1  = 200        ! throat 위쪽(수렴부) 곡률반경
r2  = 50         ! throat 아래쪽(확산부) 곡률반경
a1  = 45         ! 수렴부 반각
a2  = 15         ! 확산부 반각
yt  = 260        ! 입구 높이
ye  = -1228.5    ! 출구 높이

xb = rt+r1-r1*cos(a1)        ! R200 원호와 45도 직선의 접점
yb = r1*sin(a1)
xa = xb+(yt-yb)*tan(a1)      ! 입구 안쪽 (= 327.16)
xd = rt+r2-r2*cos(a2)        ! R50 원호와 15도 직선의 접점
yd = -r2*sin(a2)
xe = xd+(yd-ye)*tan(a2)      ! 출구 안쪽 (= 477.41)

k,1,xa,yt
k,2,xb,yb
k,3,rt,0
k,4,xd,yd
k,5,xe,ye
k,11,rt+r1,0                 ! R200 중심
k,12,rt+r2,0                 ! R50 중심

l,1,2                        ! L1 : 45도 직선
larc,2,3,11,r1               ! L2 : R200 원호
larc,3,4,12,r2               ! L3 : R50 원호
l,4,5                        ! L4 : 15도 직선
kdele,11,12

! ---- 바깥면 = 안쪽면을 t 만큼 offset (진행방향 왼쪽 = 바깥쪽) ----
offln,1,t                    ! L5
o1 = ofk1_
o2 = ofk2_
offln,2,t,o2                 ! L6 (앞 선 끝점에서 시작)
o3 = ofk2_
offln,3,t,o3                 ! L7
o4 = ofk2_
offln,4,t,o4                 ! L8
o5 = ofk2_

! ---- 바깥면 양 끝을 입구(y=yt)/출구(y=ye) 높이까지 연장 ----
tx = lsx(5,0)
ty = lsy(5,0)
kmodif,o1,kx(o1)+(yt-ky(o1))*tx/ty,yt
tx = lsx(8,1)
ty = lsy(8,1)
kmodif,o5,kx(o5)+(ye-ky(o5))*tx/ty,ye

! ---- 두께 방향 선 ----
l,1,o1                       ! L9  : 입구 끝면
l,2,o2                       ! L10
l,3,o3                       ! L11
l,4,o4                       ! L12
l,5,o5                       ! L13 : 출구 끝면
lplot

! ==================================================================
!  (c) 단면 area (4변 area 4개 -> mapped mesh 가능)
! ==================================================================
al,1,10,5,9
al,2,11,6,10
al,3,12,7,11
al,4,13,8,12
aplot

! ==================================================================
!  (d) mapped mesh
! ==================================================================
et,1,plane55
keyopt,1,3,1                 ! 축대칭
mp,kxx,1,1.143e-3            ! W/mm.K

nt = 6                       ! 두께 방향 분할 (5 mm)
lsel,s,line,,9,13
lesize,all,,,nt
lsel,all
lesize,1,,,34                ! 마주보는 변은 분할 수가 같아야 mapped 가능
lesize,5,,,34
lesize,2,,,30
lesize,6,,,30
lesize,3,,,4
lesize,7,,,4
lesize,4,,,150
lesize,8,,,150
mshape,0,2d
mshkey,1
amesh,all
eplot

! ==================================================================
!  (e) 정상상태 열해석
! ==================================================================
lsel,s,line,,1,4             ! 안쪽면 : 연소가스
sfl,all,conv,84e-6,,1073.15
lsel,s,line,,5,8             ! 바깥면 : 공기
sfl,all,conv,30e-6,,298.15
lsel,all

/solu
antype,static
solve
finish

/post1
set,last
plnsol,temp
*get,tmax,plnsol,0,max
*get,tmin,plnsol,0,min
finish

! ==================================================================
!  (f) 열응력 해석
! ==================================================================
/prep7
etchg,tts                    ! PLANE55 -> PLANE182
keyopt,1,3,1                 ! 축대칭 다시 지정
mp,ex,1,208e3                ! MPa (N/mm^2)
mp,nuxy,1,0.29
mp,alpx,1,1.28e-5
sfldele,all,all              ! 열해석용 대류 하중 삭제
sfedele,all,all,all
tref,298.15                  ! 무응력 기준온도
ldread,temp,last,,,,,rth     ! 열해석 결과(jobname.rth) 온도를 하중으로
dl,9,,uy,0                   ! 입구 끝면 축방향 고정 (반지름 방향은 자유)

/solu
antype,static
solve
finish

/post1
set,last
plnsol,s,eqv
*get,smax,plnsol,0,max
nsort,s,eqv
*get,nmax,sort,,imax
xsmax = nx(nmax)
ysmax = ny(nmax)
*status,tmax
*status,tmin
*status,smax
*status,xsmax
*status,ysmax
! /expand,36,axis,,,10       ! 3D로 돌려 보기 (선택)
