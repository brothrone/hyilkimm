! ==================================================================
!  Term Project : Thermal Stress Analysis of a Rocket Nozzle
!  준비 : offln.mac 을 ANSYS 작업 폴더에 저장해 둘 것
!  단위 : mm, W, K, MPa
!  축대칭 : Y축 = 노즐 중심축,  X = 반지름,  Y=0 = throat 높이
!
!  [안쪽면 점 손계산]   cos45 = sin45 = 0.707107, sin15 = 0.258819, tan15 = 0.267949
!   R200 중심 (350, 0),  R50 중심 (200, 0)
!   B = (350-200*cos45, 200*sin45)        = (208.578644, 141.421356)
!   A = B 에서 45도로 y=260 까지 올라간 점   = (327.157288, 260)
!   D = (200-50*cos15, -50*sin15)         = (151.703709, -12.940952)
!   E = D 에서 15도로 y=-1228.5 까지 내려간 점 = (477.411774, -1228.5)
!
!  [offset 시작/끝점을 늘린 이유]
!   45도 선을 바깥으로 30 옮기면 끝이 30*sin45 = 21.213203 만큼 내려가서 y=260 에 못 미침
!   -> 안쪽 선을 위로 21.213203 늘린 점 (348.370491, 281.213203) 에서 offset
!      => 바깥선 끝이 정확히 (369.583694, 260)
!   15도 선은 끝이 30*sin15 = 7.764571 만큼 올라감
!   -> 안쪽 선을 아래로 7.764571 늘린 점 (479.492285, -1236.264571) 까지 offset
!      => 바깥선 끝이 정확히 (508.470059, -1228.5)
! ==================================================================
/clear
/prep7
/pnum,line,1
/pnum,area,1

! ==================================================================
!  (a)+(b) wireframe
! ==================================================================
! (a) 매크로로 바깥 직선 만들기 : offln, 1(직선), x1, y1, x2, y2, 거리, kp1, kp2
offln,1,348.370491,281.213203,208.578644,141.421356,30,3,4      ! L1 : 바깥 45도 (kp3=A'', kp4=B')
offln,1,151.703709,-12.940952,479.492285,-1236.264571,30,7,8    ! L2 : 바깥 15도 (kp7=D', kp8=E'')

! ---- [영역 1] 45도 직선 부분 ----
k,1,327.157288,260           ! A
k,2,208.578644,141.421356    ! B
l,1,2                        ! L3 : 안쪽 45도 직선
l,2,4                        ! L4 : 두께 방향 선 (B-B')
l,1,3                        ! L5 : 입구 끝면 (A-A'')
al,1,3,4,5                   ! A1

! ---- [영역 2] 15도 직선 부분 ----
k,5,151.703709,-12.940952    ! D
k,6,477.411774,-1228.5       ! E
l,5,6                        ! L6 : 안쪽 15도 직선
l,5,7                        ! L7 : 두께 방향 선 (D-D')
l,6,8                        ! L8 : 출구 끝면 (E-E'')
al,2,6,7,8                   ! A2

! ---- [영역 3, 4] 원호 부분 : offln, 2(원호), 중심x, 중심y, 반지름, 시작각, 끝각, 거리 ----
offln,2,350,0,200,135,180,30  ! A3 : R200 원호 + offset R170 원호
offln,2,200,0,50,180,195,30   ! A4 : R50  원호 + offset R20  원호

nummrg,kp                    ! 이음매에 겹친 keypoint / 선 합치기
lplot

! ==================================================================
!  (c) 단면 area
! ==================================================================
aplot                        ! A1 ~ A4 (모두 4변 area)

! ==================================================================
!  (d) mapped mesh
! ==================================================================
et,1,plane55
keyopt,1,3,1                 ! 축대칭
mp,kxx,1,1.143e-3            ! W/mm.K

esize,5                      ! 요소 크기 5 mm (두께 방향 6개)
mshkey,1                     ! mapped mesh
amesh,all
eplot

! ==================================================================
!  (e) 정상상태 열해석
! ==================================================================
! ---- 직선 부분 : 선 번호를 알고 있으므로 sfl ----
sfl,3,conv,84e-6,,1073.15    ! 안쪽 45도
sfl,6,conv,84e-6,,1073.15    ! 안쪽 15도
sfl,1,conv,30e-6,,298.15     ! 바깥 45도
sfl,2,conv,30e-6,,298.15     ! 바깥 15도

! ---- 원호 부분 : pcirc 가 만든 선 번호를 모르므로 노드를 골라서 sf ----
local,11,1,350,0             ! R200 중심의 원통좌표계 (x = 중심에서의 거리)
nsel,s,loc,x,199.99,200.01   ! 거리 200 인 노드 (안쪽면)
csys,0
nsel,r,loc,y,-0.01,141.43    ! 그 중 원호 구간만
sf,all,conv,84e-6,1073.15
csys,11
nsel,s,loc,x,169.99,170.01   ! 거리 170 인 노드 (바깥면)
csys,0
nsel,r,loc,y,-0.01,141.43
sf,all,conv,30e-6,298.15

local,12,1,200,0             ! R50 중심의 원통좌표계
nsel,s,loc,x,49.99,50.01     ! 거리 50 (안쪽면)
csys,0
nsel,r,loc,y,-12.95,0.01
sf,all,conv,84e-6,1073.15
csys,12
nsel,s,loc,x,19.99,20.01     ! 거리 20 (바깥면)
csys,0
nsel,r,loc,y,-12.95,0.01
sf,all,conv,30e-6,298.15
allsel

/solu
solve

/post1
set
plnsol,temp                  ! (e) 온도 분포

! ==================================================================
!  (f) 열응력 해석
! ==================================================================
/prep7
etchg,tts                    ! PLANE55 -> PLANE182
keyopt,1,3,1                 ! etchg 하면 keyopt 가 초기화되므로 축대칭 다시 지정
mp,ex,1,208e3                ! MPa (= N/mm^2)
mp,nuxy,1,0.29
mp,alpx,1,1.28e-5

ldread,temp,,,,,file,rth     ! 열해석 결과 온도 불러오기
tref,298.15                  ! 응력이 0 인 기준 온도

nsel,s,loc,y,259.99,260.01   ! 입구 끝면 노드
d,all,uy,0                   ! 축방향만 고정 (반지름 방향은 자유)
allsel

/solu
solve

/post1
set
plnsol,s,eqv                 ! (f) 등가응력 (범례의 MX 가 최대값)
