! ==================================================================
!  Term Project : Thermal Stress Analysis of a Rocket Nozzle
!  준비 : offln.mac 을 ANSYS 작업 폴더에 저장해 둘 것
!  단위 : mm, W, K, MPa
!  축대칭 : Y축 = 노즐 중심축,  X = 반지름,  Y=0 = throat 높이
! ==================================================================
/clear
/prep7
/pnum,line,1
/pnum,area,1

*afun,deg
t   = 30                     ! 벽 두께
rt  = 150                    ! throat 안쪽 반지름
r1  = 200                    ! throat 위쪽 곡률반경
r2  = 50                     ! throat 아래쪽 곡률반경
a1  = 45                     ! 수렴부 각도
a2  = 15                     ! 확산부 각도
yt  = 260                    ! 입구 높이
ye  = -1228.5                ! 출구 높이

! ---- 안쪽면 주요 점 ----
xb = rt+r1-r1*cos(a1)        ! B : 45도 직선과 R200 이 만나는 점
yb = r1*sin(a1)
xa = xb+(yt-yb)*tan(a1)      ! A : 입구 안쪽 (327.16, 260)
xd = rt+r2-r2*cos(a2)        ! D : R50 과 15도 직선이 만나는 점
yd = -r2*sin(a2)
xe = xd+(yd-ye)*tan(a2)      ! E : 출구 안쪽 (477.41, -1228.5)

! ==================================================================
!  (a)+(b) wireframe
! ==================================================================
dist = t

! ---- [영역 1] 45도 직선 부분 ----
! offset 하면 끝점이 아래로 t*sin(45) 내려가므로,
! 안쪽 선을 그만큼 위로 늘린 점(xs,ys)에서 offset 해서 바깥 끝이 정확히 y=260 에 오게 함
ys = yt+t*sin(a1)
xs = xb+(ys-yb)*tan(a1)
x1 = xs
y1 = ys
x2 = xb
y2 = yb
kp1 = 3
kp2 = 4
offln                        ! L1 : 바깥 45도 직선 (kp3 -> kp4)
k,1,xa,yt
k,2,xb,yb
l,1,2                        ! L2 : 안쪽 45도 직선
l,2,4                        ! L3 : 두께 방향 선
l,1,3                        ! L4 : 입구 끝면
al,1,2,3,4                   ! A1

! ---- [영역 2] 15도 직선 부분 ----
! 같은 방법 : 출구 쪽으로 t*sin(15) 만큼 늘린 점에서 offset
yes = ye-t*sin(a2)
xes = xd+(yd-yes)*tan(a2)
x1 = xd
y1 = yd
x2 = xes
y2 = yes
kp1 = 7
kp2 = 8
offln                        ! L5 : 바깥 15도 직선 (kp7 -> kp8)
k,5,xd,yd
k,6,xe,ye
l,5,6                        ! L6 : 안쪽 15도 직선
l,5,7                        ! L7 : 두께 방향 선
l,6,8                        ! L8 : 출구 끝면
al,5,6,7,8                   ! A2

! ---- [영역 3, 4] 원호 부분 : 두께 t 인 부채꼴 고리 ----
wpave,rt+r1,0                ! R200 중심으로 working plane 이동
pcirc,r1-t,r1,135,180        ! A3 : R170 ~ R200, 135 ~ 180도
wpave,rt+r2,0                ! R50 중심으로 이동
pcirc,r2-t,r2,180,195        ! A4 : R20 ~ R50, 180 ~ 195도
wpave,0,0

nummrg,kp                    ! 이음매에 겹친 keypoint / 선 합치기
lplot

! ==================================================================
!  (c) 단면 area
! ==================================================================
aplot                        ! A1 ~ A4 (모두 4변 area)

! ==================================================================
!  (d) mapped mesh
! ==================================================================
et,1,plane55
keyopt,1,3,1                 ! 축대칭
mp,kxx,1,1.143e-3            ! W/mm.K

esize,5                      ! 요소 크기 5 mm (두께 방향 6개)
mshkey,1                     ! mapped mesh
amesh,all
eplot

! ==================================================================
!  (e) 정상상태 열해석
! ==================================================================
! ---- 직선 부분 : 선 번호를 알고 있으므로 sfl ----
sfl,2,conv,84e-6,,1073.15    ! 안쪽 45도
sfl,6,conv,84e-6,,1073.15    ! 안쪽 15도
sfl,1,conv,30e-6,,298.15     ! 바깥 45도
sfl,5,conv,30e-6,,298.15     ! 바깥 15도

! ---- 원호 부분 : pcirc 가 만든 선 번호를 모르므로 노드를 골라서 sf ----
local,11,1,rt+r1,0           ! R200 중심의 원통좌표계 (x = 반지름)
nsel,s,loc,x,r1-0.01,r1+0.01         ! 반지름 200 인 노드
csys,0
nsel,r,loc,y,-0.01,yb+0.01           ! 그 중 원호 구간만
sf,all,conv,84e-6,1073.15
csys,11
nsel,s,loc,x,r1-t-0.01,r1-t+0.01     ! 반지름 170 인 노드
csys,0
nsel,r,loc,y,-0.01,yb+0.01
sf,all,conv,30e-6,298.15

local,12,1,rt+r2,0           ! R50 중심의 원통좌표계
nsel,s,loc,x,r2-0.01,r2+0.01         ! 반지름 50
csys,0
nsel,r,loc,y,yd-0.01,0.01
sf,all,conv,84e-6,1073.15
csys,12
nsel,s,loc,x,r2-t-0.01,r2-t+0.01     ! 반지름 20
csys,0
nsel,r,loc,y,yd-0.01,0.01
sf,all,conv,30e-6,298.15
allsel

/solu
solve

/post1
set
plnsol,temp                  ! (e) 온도 분포

! ==================================================================
!  (f) 열응력 해석
! ==================================================================
/prep7
etchg,tts                    ! PLANE55 -> PLANE182
keyopt,1,3,1                 ! etchg 하면 keyopt 가 초기화되므로 축대칭 다시 지정
mp,ex,1,208e3                ! MPa (= N/mm^2)
mp,nuxy,1,0.29
mp,alpx,1,1.28e-5

ldread,temp,,,,,file,rth     ! 열해석 결과 온도 불러오기
tref,298.15                  ! 응력이 0 인 기준 온도

nsel,s,loc,y,yt-0.01,yt+0.01 ! 입구 끝면 노드
d,all,uy,0                   ! 축방향만 고정 (반지름 방향은 자유)
allsel

/solu
solve

/post1
set
plnsol,s,eqv                 ! (f) 등가응력 (범례의 MX 가 최대값)
